Quickstart

Up in under five minutes

bashinstall · configure · run
# clone & bootstrap
git clone https://github.com/Mourad-Soltani/free-ai-gateway
cd free-ai-gateway && ./install.sh

# add keys + generate master key
nano .env
python -c "import secrets; print('sk-' + secrets.token_urlsafe(32))"

# launch proxy on :4000
./run.sh

# query or run the agent
python client.py "Explain SQLite WAL in two bullets"
python agents/agent.py --task "Summarise today's free-tier status"
